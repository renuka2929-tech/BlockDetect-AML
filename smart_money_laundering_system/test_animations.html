Test 1: Check if animation elements exist

const elements = document.querySelectorAll('[class*="scroll-"]');
console.log(`Found ${elements.length} animated elements`);
elements.forEach((el, i) => {
    const classes = Array.from(el.classList).filter(c => c.includes('scroll-'));
    console.log(`${i+1}. ${el.tagName} - ${classes.join(', ')} - Revealed: ${el.classList.contains('revealed')}`);
});

Test 2: Check revealed status and visibility

const animated = document.querySelectorAll('[class*="scroll-"]');
const revealed = document.querySelectorAll('[class*="scroll-"].revealed');
const notRevealed = Array.from(animated).filter(el => !el.classList.contains('revealed'));
console.log(`Total: ${animated.length}, Revealed: ${revealed.length}, Not Revealed: ${notRevealed.length}`);
notRevealed.forEach(el => {
    const rect = el.getBoundingClientRect();
    const isVisible = rect.top < window.innerHeight && rect.bottom > 0;
    console.log('Not revealed but visible?', isVisible, el);
});

Test 3: Force reveal all visible elements

const elements = document.querySelectorAll('[class*="scroll-"]');
let count = 0;
elements.forEach(el => {
    const rect = el.getBoundingClientRect();
    const isVisible = rect.top < window.innerHeight && rect.bottom > 0;
    if (isVisible && !el.classList.contains('revealed')) {
        el.classList.add('revealed');
        count++;
    }
});
console.log(`Force revealed ${count} elements`);

Test 5: Check element opacity

const elements = document.querySelectorAll('[class*="scroll-"]');
elements.forEach((el, i) => {
    const style = window.getComputedStyle(el);
    const opacity = style.opacity;
    const display = style.display;
    const visibility = style.visibility;
    if (opacity < 0.5 || display === 'none' || visibility === 'hidden') {
        console.warn(`Element ${i+1} is hidden:`, {opacity, display, visibility}, el);
    }
});
